KONFIGURASI DATABASE SUPABASE

Status Aktif: Kredensial proyek Supabase Anda sudah tertanam secara permanen dalam mode produksi sehingga setiap perangkat yang membuka link ini otomatis terhubung ke Cloud PostgreSQL.
SKRIP SQL TABEL SUPABASE (REFERENSI) 
-- Skrip SQL Tabel SIPENJASKES Supabase:
CREATE TABLE IF NOT EXISTS profil_sekolah (
  id TEXT PRIMARY KEY DEFAULT 'default',
  nama_sekolah TEXT DEFAULT '',
  alamat_sekolah TEXT DEFAULT '',
  nama_kepsek TEXT DEFAULT '',
  nip_kepsek TEXT DEFAULT '',
  nama_guru TEXT DEFAULT '',
  nip_guru TEXT DEFAULT '',
  tahun_ajaran TEXT DEFAULT '',
  semester TEXT DEFAULT 'Ganjil',
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS rombel (
  id TEXT PRIMARY KEY,
  tingkat TEXT NOT NULL,
  paralel TEXT DEFAULT '',
  nama_rombel TEXT NOT NULL,
  wali_kelas TEXT DEFAULT '',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS siswa (
  id TEXT PRIMARY KEY,
  rombel_id TEXT NOT NULL,
  nis TEXT DEFAULT '',
  nama TEXT NOT NULL,
  jk TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS roster (
  id TEXT PRIMARY KEY,
  hari TEXT NOT NULL,
  rombel_id TEXT NOT NULL,
  jp INTEGER DEFAULT 3,
  jam_mulai TEXT NOT NULL,
  jam_selesai TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS absensi (
  id TEXT PRIMARY KEY,
  tanggal TEXT NOT NULL,
  rombel_id TEXT NOT NULL,
  siswa_id TEXT NOT NULL,
  status TEXT DEFAULT 'H',
  keterangan TEXT DEFAULT '',
  materi TEXT DEFAULT '',
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS cp_tp_kktp (
  id TEXT PRIMARY KEY,
  fase TEXT NOT NULL,
  elemen TEXT NOT NULL,
  tp TEXT NOT NULL,
  atp TEXT DEFAULT '',
  kktp INTEGER DEFAULT 75,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS penilaian (
  id TEXT PRIMARY KEY,
  rombel_id TEXT NOT NULL,
  siswa_id TEXT NOT NULL,
  tp_label TEXT DEFAULT 'Umum',
  diagnostik NUMERIC DEFAULT 0,
  formatif NUMERIC DEFAULT 0,
  sumatif NUMERIC DEFAULT 0,
  nilai_akhir NUMERIC DEFAULT 0,
  kktp_batas INTEGER DEFAULT 75,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
